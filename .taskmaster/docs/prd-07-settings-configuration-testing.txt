## Test Scenarios

### Settings Update
```typescript
test('should update general settings', async ({ authenticatedPage }) => {
  await authenticatedPage.goto('/pi-controller/settings');

  // Update hostname
  await page.fill('input[name="hostname"]', 'new-controller-name');

  // Update timezone
  await page.selectOption('select[name="timezone"]', 'America/New_York');

  // Save changes
  await page.click('button:has-text("Save Changes")');

  // Verify success
  await expect(page.locator('.success-toast')).toBeVisible();

  // Reload and verify persistence
  await page.reload();
  await expect(page.locator('input[name="hostname"]')).toHaveValue('new-controller-name');
});
```

### YAML Export/Import
```typescript
test('should export configuration as YAML', async ({ authenticatedPage }) => {
  await authenticatedPage.goto('/pi-controller/settings');

  const downloadPromise = page.waitForEvent('download');
  await page.click('button:has-text("Export YAML")');

  const download = await downloadPromise;
  expect(download.suggestedFilename()).toMatch(/config.*\.yaml/);

  // Verify YAML content
  const path = await download.path();
  const content = await fs.readFile(path, 'utf-8');
  expect(content).toContain('hostname:');
  expect(content).toContain('timezone:');
});
```
