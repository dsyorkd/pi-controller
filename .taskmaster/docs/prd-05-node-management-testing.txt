## Test Scenarios

### Node Filtering
```typescript
test('should filter nodes by status', async ({ authenticatedPage }) => {
  await authenticatedPage.goto('/pi-controller/nodes');

  await page.selectOption('select[name="status"]', 'online');

  const nodes = await page.locator('.node-card').all();
  for (const node of nodes) {
    const status = await node.getAttribute('data-status');
    expect(status).toBe('online');
  }
});
```

### Node Adoption
```typescript
test('should adopt discovered node', async ({ authenticatedPage }) => {
  await authenticatedPage.goto('/pi-controller/nodes');

  // Open discovery panel
  await page.click('[data-testid="discovery-panel-toggle"]');
  await expect(page.locator('.discovery-panel')).toBeVisible();

  // Adopt first discovered node
  await page.click('.discovered-node:first-child button:has-text("Adopt")');

  // Fill adoption form
  await page.fill('input[name="ssh-user"]', 'pi');
  await page.fill('input[name="ssh-password"]', 'raspberry');
  await page.click('button:has-text("Confirm")');

  // Verify success
  await expect(page.locator('.success-toast')).toBeVisible();
  await expect(page.locator('.success-toast')).toContainText(/adopted successfully/i);
});
```
