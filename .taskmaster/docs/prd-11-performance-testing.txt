## Test Scenarios

### Page Load Performance
```typescript
test('should load dashboard in under 3 seconds', async ({ authenticatedPage }) => {
  const startTime = Date.now();

  await authenticatedPage.goto('/pi-controller');
  await page.waitForLoadState('networkidle');

  const loadTime = Date.now() - startTime;
  expect(loadTime).toBeLessThan(3000);
});
```

### Core Web Vitals
```typescript
test('should meet Core Web Vitals thresholds', async ({ authenticatedPage }) => {
  await authenticatedPage.goto('/pi-controller');

  const metrics = await page.evaluate(() => {
    return new Promise(resolve => {
      new PerformanceObserver(list => {
        const entries = list.getEntries();
        const lcp = entries.find(e => e.entryType === 'largest-contentful-paint');
        resolve({ lcp: lcp?.renderTime });
      }).observe({ entryTypes: ['largest-contentful-paint'] });
    });
  });

  expect(metrics.lcp).toBeLessThan(2500); // 2.5 seconds
});
```
